     public static Integer subtractSummation( BTNode root ){
        Integer sum=sumOfTree(root.left)-sumOfTree(root.right);
        return sum;
    }


    private static Integer sumOfTree(BTNode root) {
        if(root==null) return 0;
        return (Integer)root.elem+sumOfTree(root.left)+sumOfTree(root.right);
    }

 public static void swapChild( BTNode root, int lvl, int M ){
        if(root==null || lvl==M) return;


        BTNode temp=root.left;
        root.left=root.right;
        root.right=temp;


        swapChild(root.left,lvl+1,M);
        swapChild(root.right,lvl+1,M);
    }

   public static Integer rangeSum( BSTNode root, Integer low, Integer high ){
        if(root==null) return 0;
        Integer x=root.elem;
        if(root.elem<low || root.elem>high) x=0;
        return x+rangeSum(root.left,low,high)+rangeSum(root.right,low,high);
    }

     public static Integer mirrorSum( BSTNode root ){


       if (root == null) return 0;
        return mirrorSumHelper(root.left,root.right);
    }


    private static int mirrorSumHelper(BSTNode l,BSTNode r){


        if (l==null|| r==null) {
            return 0;
        }
        int curr=(Integer)l.elem+(Integer)r.elem;
        int outSum=mirrorSumHelper(l.left, r.right);
        int inSum=mirrorSumHelper(l.right, r.left);


        return curr+outSum+inSum;
    }

  public static Boolean isBST( BSTNode root ){
       
        if(root==null) return true;
        if(root.left!=null && root.left.elem>root.elem) return false;
        if(root.right!=null && root.right.elem<root.elem) return false;


        return isBST(root.left) && isBST(root.right);
    }
