 public void remove( Integer key ){


        Integer idx=hashFunction(key);


        if((idx==-1) || (ht[idx]==null)) return;
       
        PairNode pred=ht[idx];
        PairNode n=ht[idx].next;    




        if(pred.key==key){
            ht[idx]=pred.next;
            return;
        }
        while(n!=null){
            if(n.key==key){
                pred.next=n.next;
                return;
            }
            n=n.next;
            pred=pred.next;
        }


    }

 private int hashFunction( String key ){
       
        if(key==null) return -1; //remove this line


        String temp=key;


        if(key.length()%2!=0) temp+='N';


        int hash = 0;


        for(int i=0; i<temp.length(); i+=2){
            int a = temp.charAt(i);
            int b = temp.charAt(i+1);
            hash += Integer.parseInt(String.valueOf(a) + String.valueOf(b));
        }


        hash = hash%ht.length;
        return hash;
    }




 public String searchHashtable( Object[] keyValuePair ){


        String key = (String)keyValuePair[0];
        Integer value = (Integer)keyValuePair[1];
        int idx = hashFunction(key);


        if(idx==-1 || ht[idx]==null) return "Not Found";


        PairNode pNode = ht[idx];


        while(pNode!=null){
            if(pNode.key.equals(key) && pNode.value.equals(value)) return "Found";
            pNode = pNode.next;
        }


        return "Not Found";
    }

   public static Stack conditionalReverse(Stack stack) {
        Stack rStack = new Stack();
       
        int x=0;


        while(!stack.isEmpty()) {


            if(rStack.isEmpty()){
                x = stack.pop();
                rStack.push(x);
                continue;
            }


            int curr = stack.pop();


            if(x==curr) continue;
           
            x=curr;
            rStack.push(x);
        }
        return rStack;
    }

    public static String hotPotato(String[] players, int k) {


        LinkedListQueue q=new LinkedListQueue();


        for(int i=0;i<players.length;i++){
            q.enqueue(players[i]);
        }


        int count=0;
        Object p="";


        while(true){


            p=q.dequeue();


            if(q.isEmpty()) break;


            if(count%k==0 && count!=0){
                System.out.println(p+" eliminated");
                count=0;
                continue;
            }
            q.enqueue(p);
            count++;
        }


        return (String)p;
    }
