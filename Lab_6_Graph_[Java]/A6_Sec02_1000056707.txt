class EdgeNode {
    int toV;
    int weight;
    EdgeNode next;


    public EdgeNode(int toV, int weight) {
        this.toV = toV;
        this.weight = weight;
        this.next = null;
    }
}


// GraphUtils.java
class GraphUtils {
    
    public static void printLL(EdgeNode head) {
        EdgeNode n = head;
        while (n != null) {
            System.out.print(" -> (" + n.toV + "," + n.weight + ")");
            n = n.next;
        }
        System.out.println();
    }


    // Append a new edge node to adjacency list
    public static void appendLL(EdgeNode head, EdgeNode eNode) {
        EdgeNode n = head;
        while (n.next != null) {
            n = n.next;
        }
        n.next = eNode;
    }


    
    public static void addEdge(EdgeNode[] adjList, int from, int to, int weight) {
        EdgeNode newNode = new EdgeNode(to, weight);
        if (adjList[from] == null) {
            adjList[from] = newNode;
        } else {
            GraphUtils.appendLL(adjList[from], newNode);
        }
    }


    
    public static void showAdjList(EdgeNode[] adjList) {
        for (int i = 0; i < adjList.length; i++) {
            System.out.print("Vertex " + i + ":");
            printLL(adjList[i]);
        }
    }
}

public static int maxWeight(int[][] obj){


        int temp=0;
        int max=0;


        for(int i=0; i<obj.length; i++){
            temp=0;
            for(int j=0; j<obj[0].length;j++){
                if(obj[i][j]>0){
                    temp+=obj[i][j];
                }
            }
            if(temp>max) max=temp;
        }
        return max;
    }

    public static void directedToUndirected(int[][] obj){


        for(int i=0; i<obj.length; i++){
            for(int j=i+1; j<obj[0].length;j++){
                if(obj[i][j]>0 || obj[j][i]>0){
                    int temp=obj[i][j]+obj[j][i];
                    obj[j][i]=obj[i][j]=temp;
                }
            }
        }
    }

  public static void directedToUndirected(EdgeNode[] obj){

        EdgeNode t=null;

        for(int i=0; i<obj.length;i++){
            t=obj[i];
            EdgeNode x=null;

            while (t!=null){
                x=obj[t.toV];
                boolean twoWays=false;
                while(x!=null){

                    if(x.toV==i){
                        twoWays=true;
                        break;
                    }
                    x=x.next;
                }
                if(twoWays){
                    if(i < t.toV){
                        int sum=t.weight+x.weight;
                        t.weight=sum;
                        x.weight=sum;
                    }
                }
                else GraphUtils.addEdge(obj,t.toV,i,t.weight);
                t=t.next;
            }
        }
